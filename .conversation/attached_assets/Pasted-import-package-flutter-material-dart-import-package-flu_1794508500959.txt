import 'package:flutter/material.dart';
import 'package:flutter_inappwebview/flutter_inappwebview.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'dart:convert';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  runApp(const IkramApp());
}

class IkramApp extends StatelessWidget {
  const IkramApp({Key? key}) : super(key: key);

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'إكرام',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        primarySwatch: Colors.pink,
        scaffoldBackgroundColor: const Color(0xFFFFF0F5), // لون خلفية روزي خفيف (LavenderBlush/Pink tint)
        appBarTheme: const AppBarTheme(
          backgroundColor: Colors.pinkAccent,
          foregroundColor: Colors.white,
        ),
      ),
      home: const ProfilesScreen(),
    );
  }
}

class Profile {
  String id;
  String name;
  String proxy; // Format: ip:port or user:pass@ip:port

  Profile({required this.id, required this.name, required this.proxy});

  Map<String, dynamic> toJson() => {'id': id, 'name': name, 'proxy': proxy};
  factory Profile.fromJson(Map<String, dynamic> json) => Profile(
        id: json['id'],
        name: json['name'],
        proxy: json['proxy'] ?? '',
      );
}

class ProfilesScreen extends StatefulWidget {
  const ProfilesScreen({Key? key}) : super(key: key);

  @override
  _ProfilesScreenState createState() => _ProfilesScreenState();
}

class _ProfilesScreenState extends State<ProfilesScreen> {
  List<Profile> profiles = [];

  @override
  void initState() {
    super.initState();
    _loadProfiles();
  }

  Future<void> _loadProfiles() async {
    final prefs = await SharedPreferences.getInstance();
    final String? profilesString = prefs.getString('ikram_profiles');
    if (profilesString != null) {
      List decoded = jsonDecode(profilesString);
      setState(() {
        profiles = decoded.map((e) => Profile.fromJson(e)).toList();
      });
    } else {
      // بروفايل افتراضي أول
      setState(() {
        profiles.add(Profile(id: '1', name: 'بروفايل 1 - شي إن', proxy: ''));
      });
      _saveProfiles();
    }
  }

  Future<void> _saveProfiles() async {
    final prefs = await SharedPreferences.getInstance();
    prefs.setString('ikram_profiles', jsonEncode(profiles.map((e) => e.toJson()).toList()));
  }

  void _addProfile() {
    setState(() {
      String newId = DateTime.now().millisecondsSinceEpoch.toString();
      profiles.add(Profile(id: newId, name: 'بروفايل ${profiles.length + 1}', proxy: ''));
      _saveProfiles();
    });
  }

  void _deleteProfile(String id) async {
    // مسح الكوكيز والبيانات الخاصة بهذا البروفايل بالكامل لضمان عدم التتبع
    CookieManager cookieManager = CookieManager.instance();
    await cookieManager.deleteAllCookies();
    
    // مسح الـ WebView cache لهذا البروفايل
    InAppWebViewController.clearAllCache();

    setState(() {
      profiles.removeWhere((p) => p.id == id);
      _saveProfiles();
    });

    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('تم مسح الكوكيز والبيانات بنجاح وتطهير البروفايل!')),
    );
  }

  // زر لمسح جميع الكوكيز والبيانات لكل البروفايلات دفعة واحدة
  Future<void> _clearAllData() async {
    CookieManager cookieManager = CookieManager.instance();
    await cookieManager.deleteAllCookies();
    await InAppWebViewController.clearAllCache();
    
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('تم مسح جميع الكوكيز والبيانات بنجاح!')),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('متصفح إكرام للخصوصية'),
        actions: [
          IconButton(
            icon: const Icon(Icons.delete_sweep),
            tooltip: 'مسح الكوكيز الشامل',
            onPressed: _clearAllData,
          ),
        ],
      ),
      body: ListView.builder(
        itemCount: profiles.length,
        itemBuilder: (context, index) {
          final profile = profiles[index];
          return Card(
            margin: const EdgeInsets.all(8.0),
            color: Colors.white,
            elevation: 3,
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
            child: ListTile(
              title: Text(
                profile.name,
                style: const TextStyle(fontWeight: FontWeight.bold, color: Colors.pinkAccent),
              ),
              subtitle: Text(
                profile.proxy.isEmpty ? 'بدون بروكسي (IP حقيقي)' : 'البروكسي: ${profile.proxy}',
                style: TextStyle(color: Colors.grey[600]),
              ),
              trailing: Row(
                mainAxisSize: MainAxisSize.min,
                children: [
                  IconButton(
                    icon: const Icon(Icons.edit, color: Colors.pink),
                    onPressed: () => _editProfileDialog(profile),
                  ),
                  IconButton(
                    icon: const Icon(Icons.delete, color: Colors.redAccent),
                    onPressed: () => _deleteProfile(profile.id),
                  ),
                ],
              ),
              onTap: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(
                    builder: (context) => BrowserScreen(profile: profile),
                  ),
                );
              },
            ),
          );
        },
      ),
      floatingActionButton: FloatingActionButton(
        backgroundColor: Colors.pinkAccent,
        onPressed: _addProfile,
        child: const Icon(Icons.add, color: Colors.white),
      ),
    );
  }

  void _editProfileDialog(Profile profile) {
    TextEditingController nameController = TextEditingController(text: profile.name);
    TextEditingController proxyController = TextEditingController(text: profile.proxy);

    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: const Text('إعدادات البروفايل والبروكسي'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              TextField(
                controller: nameController,
                decoration: const InputDecoration(labelText: 'اسم البروفايل'),
              ),
              const SizedBox(height: 10),
              TextField(
                controller: proxyController,
                decoration: const InputDecoration(
                  labelText: 'البروكسي (مثال: IP:Port أو User:Pass@IP:Port)',
                ),
              ),
            ],
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('إلغاء'),
            ),
            ElevatedButton(
              style: ElevatedButton.styleFrom(backgroundColor: Colors.pinkAccent),
              onPressed: () {
                setState(() {
                  profile.name = nameController.text;
                  profile.proxy = proxyController.text;
                  _saveProfiles();
                });
                Navigator.pop(context);
              },
              child: const Text('حفظ', style: TextStyle(color: Colors.white)),
            ),
          ],
        );
      },
    );
  }
}

class BrowserScreen extends StatefulWidget {
  final Profile profile;
  const BrowserScreen({Key? key, required this.profile}) : super(key: key);

  @override
  _BrowserScreenState createState() => _BrowserScreenState();
}

class _BrowserScreenState extends State<BrowserScreen> {
  InAppWebViewController? webViewController;
  TextEditingController urlController = TextEditingController(text: 'https://www.shein.com');
  double progress = 0;
  String currentProxyTop = '';

  @override
  void initState() {
    super.initState();
    currentProxyTop = widget.profile.proxy;
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(widget.profile.name),
      ),
      body: Column(
        children: [
          // شريط البروكسي في الفوق وطريقة التحكم فيه
          Container(
            padding: const EdgeInsets.all(8.0),
            color: Colors.pink[50],
            child: Row(
              children: [
                const Icon(Icons.security, size: 18, color: Colors.pinkAccent),
                const SizedBox(width: 8),
                Expanded(
                  child: Text(
                    currentProxyTop.isEmpty ? 'البروكسي الفوق: غير مُفعّل (مباشر)' : 'البروكسي الفوق: $currentProxyTop',
                    style: const TextStyle(fontSize: 12, fontWeight: FontWeight.bold, color: Colors.black87),
                    overflow: TextOverflow.ellipsis,
                  ),
                ),
                TextButton.icon(
                  onPressed: () {
                    _showProxyTopDialog();
                  },
                  icon: const Icon(Icons.settings, size: 14, color: Colors.pink),
                  label: const Text('تغيير البروكسي', style: TextStyle(fontSize: 12, color: Colors.pink)),
                ),
              ],
            ),
          ),
          // شريط العنوان والبحث
          Container(
            padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 4),
            color: Colors.white,
            child: Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: urlController,
                    decoration: const InputDecoration(
                      hintText: 'أدخل رابط الموقع...',
                      border: InputBorder.none,
                    ),
                    onSubmitted: (value) {
                      var url = Uri.parse(value);
                      if (!url.hasScheme) {
                        url = Uri.parse('https://www.google.com/search?q=$value');
                      }
                      webViewController?.loadUrl(urlRequest: URLRequest(url: WebUri(url.toString())));
                    },
                  ),
                ),
                IconButton(
                  icon: const Icon(Icons.refresh, color: Colors.pink),
                  onPressed: () => webViewController?.reload(),
                ),
              ],
            ),
          ),
          if (progress < 1.0)
            LinearProgressIndicator(value: progress, color: Colors.pinkAccent),
          
          // متصفح الويب المعزول
          Expanded(
            child: InAppWebView(
              initialUrlRequest: URLRequest(url: WebUri('https://www.shein.com')),
              initialSettings: InAppWebViewSettings(
                javaScriptEnabled: true,
                clearCache: false,
                // ميزات منع التتبع وحماية الخصوصية
                supportMultipleWindows: false,
                databaseEnabled: true,
                domStorageEnabled: true,
              ),
              onWebViewCreated: (controller) {
                webViewController = controller;
              },
              onProgressChanged: (controller, p) {
                setState(() {
                  progress = p / 100;
                });
              },
              onLoadStop: (controller, url) async {
                urlController.text = url.toString();
              },
            ),
          ),
        ],
      ),
    );
  }

  void _showProxyTopDialog() {
    TextEditingController dialogProxyController = TextEditingController(text: currentProxyTop);
    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: const Text('تعديل بروكسي الجلسة الحالية'),
          content: TextField(
            controller: dialogProxyController,
            decoration: const InputDecoration(labelText: 'IP:Port أو User:Pass@IP:Port'),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('إلغاء'),
            ),
            ElevatedButton(
              style: ElevatedButton.styleFrom(backgroundColor: Colors.pinkAccent),
              onPressed: () {
                setState(() {
                  currentProxyTop = dialogProxyController.text;
                  widget.profile.proxy = currentProxyTop;
                });
                Navigator.pop(context);
                ScaffoldMessenger.of(context).showSnackBar(
                  const SnackBar(content: Text('تم تحديث البروكسي بنجاح!')),
                );
              },
              child: const Text('حفظ', style: TextStyle(color: Colors.white)),
            ),
          ],
        );
      },
    );
  }
}
